def fit_standardizer(X):
    mean = X.mean(axis=0)
    scale = X.std(axis=0, ddof=0)
    scale = np.where(scale == 0, 1., scale)
    return mean, scale

def apply_standardizer(X, mean, scale):
    return (X - mean) / scale

def assign_clusters(U, centres):
    # TODO 3
    # (n, 1, p) - (1, K, p) -> (n, K, p); sum over features -> (n, K)
    sq_dist = np.sum((U[:, None, :] - centres[None, :, :]) ** 2, axis=2)
    return np.argmin(sq_dist, axis=1).astype(int)  # argmin picks the first index in a tie

def update_centres(U, labels, old_centres):
    # TODO 4
    new_centres = np.array(old_centres, dtype=float, copy=True)  # never modify the input
    for k in range(len(old_centres)):
        members = U[labels == k]
        if len(members) > 0:                      # empty cluster keeps its old centre
            new_centres[k] = members.mean(axis=0)
    return new_centres

def run_kmeans(U, K=2, seed=SEED):
    rng = np.random.default_rng(seed)
    best = None
    for _ in range(10):
        centres = U[rng.choice(len(U), K, replace=False)].copy()
        for _ in range(100):
            labels = assign_clusters(U, centres)
            updated = update_centres(U, labels, centres)
            converged = np.allclose(updated, centres, rtol=0, atol=1e-8)
            centres = updated
            if converged:
                break
        labels = assign_clusters(U, centres)
        inertia = float(np.sum((U - centres[labels]) ** 2))
        if best is None or inertia < best["inertia"]:
            best = {"centres": centres.copy(), "labels": labels.copy(),
                    "inertia": inertia}
    return best

S1 (5 marks).

(a) What do the rows and two columns of radio_Z_train represent? How do these scores differ from the K-means labels?

(b) Why standardize the features before distance-based clustering?

Answer:

(a) Each row of radio_Z_train is one training candidate (1,650 rows). The two columns are its PC1 and PC2 scores: projections of its four standardized features onto the two directions of greatest variance. The scores are continuous coordinates. A K-means label is a single integer (0 or 1) saying which centroid is nearest in the full four-feature space. PCA gives a position; K-means gives a group.

(b) K-means uses squared Euclidean distance, so features with large numeric ranges would dominate it. The raw features have very different spreads: profile_std has a standard deviation of about 7, but dm_snr_mean has one of about 32, so it would dominate the distances unscaled. Standardizing gives every feature mean 0 and standard deviation 1, so each contributes on a comparable scale.

S2 (5 marks).

(a) Why is a candidate's negative log density not its probability of being a pulsar?

(b) What does the 95th-percentile threshold mean for the background calibration sample? Does it guarantee a 5% background false-alarm rate on test data?

Answer:

(a) The KDE is fitted only to background candidates. Its negative log density says how unusual a candidate is compared with the background, not how likely it is to be a pulsar. It is not a probability. It can be any real number and is not bounded in [0, 1]. A density can also exceed 1. Turning it into P(pulsar) would also need a pulsar density and the class prior (Bayes' rule).

(b) By construction, about 5% of the 400 background calibration candidates score above tau (20 of 400 are flagged). That does not guarantee a 5% false-alarm rate on test data. The test background is a different random sample, so its rate varies around 5%. The density and tau are also estimated from a finite sample, and any shift between calibration and test data would change the rate. 5% is an approximate target, not a guarantee.

def detection_summary(actual, flagged):
    actual = np.asarray(actual, dtype=int)
    flagged = np.asarray(flagged, dtype=bool)
    tn = int(np.sum((actual == 0) & ~flagged))
    fp = int(np.sum((actual == 0) & flagged))
    fn = int(np.sum((actual == 1) & ~flagged))
    tp = int(np.sum((actual == 1) & flagged))
    return {"tn": tn, "fp": fp, "fn": fn, "tp": tp,
            "precision": tp / (tp + fp) if tp + fp else 0.,
            "recall": tp / (tp + fn) if tp + fn else 0.,
            "background_false_alarm_rate": fp / (tn + fp) if tn + fp else 0.}

Answer:

(a) For a batch of 64 images (also printed by the shape check above):

  • after the first convolution: (64, 8, 28, 28) (padding 1 keeps the 28 x 28 size)
  • after the first pooling layer: (64, 8, 14, 14)
  • after the second pooling layer: (64, 16, 7, 7)
  • after the final linear layer: (64, 3), one raw score per class

(b) Each filter is one small set of weights, for example 3 x 3 for each input channel plus a bias. The same weights slide across every position in the image, and the dot product is computed at each location. So a pattern such as an edge is detected the same way wherever it appears. This is why the first conv layer has only 8 x (1 x 3 x 3) + 8 = 80 parameters, however large the image is.

Supplied training helper¶

The two losses below evaluate the full training and validation samples using the weights at the end of each epoch.

In [16]:
def cnn_loss(model, X, y):
    model.eval()
    with torch.no_grad():
        return float(nn.functional.cross_entropy(model(X), y))

def fit_cnn(model, loader, X_train, y_train, X_val, y_val):
    optimizer = torch.optim.Adam(model.parameters(), lr=0.003)
    history = {"train_ce": [], "val_ce": [], "best_epoch": None}
    best_loss, best_state = float("inf"), None
    for epoch in range(1, 11):
        model.train()
        for X_batch, y_batch in loader:
            optimizer.zero_grad()
            loss = nn.functional.cross_entropy(model(X_batch), y_batch)
            loss.backward()
            optimizer.step()
        train_ce = cnn_loss(model, X_train, y_train)
        val_ce = cnn_loss(model, X_val, y_val)
        history["train_ce"].append(train_ce)
        history["val_ce"].append(val_ce)
        if val_ce < best_loss:
            best_loss = val_ce
            best_state = copy.deepcopy(model.state_dict())
            history["best_epoch"] = epoch
    model.load_state_dict(best_state)
    return history
